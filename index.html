function num(v) {

  const n = parseFloat(String(v).replace(/[^0-9.\-]/g, ''));

  return Number.isFinite(n) ? n : null;

function parseCSV(t) {

  const rows = []; let row = [], f = '', q = false;

  for (let i = 0; i < t.length; i++) {

    const c = t[i];

    if (q) {

      if (c === '"') { if (t[i + 1] === '"') { f += '"'; i++; } else q = false; }

      else f += c;

    } else if (c === '"') q = true;

    else if (c === ',') { row.push(f); f = ''; }

    else if (c === '\n') { row.push(f); rows.push(row); row = []; f = ''; }

    else if (c !== '\r') f += c;

  }

  if (f !== '' || row.length) { row.push(f); rows.push(row); }

  return rows;

}

function toObjects(rows) {

  if (!rows.length) return [];

  const h = rows[0].map(x => x.trim());

  return rows.slice(1).filter(r => r.some(c => c.trim() !== '')).map(r => Object.fromEntries(h.map((k, i) => [k, r[i] ?? ''])));

}


function oppNick(game) {

  const m = game.match(/(?:vs|@) (.+)$/);

  if (!m) return 'Opponent';

  if (/Trail Blazers$/.test(m[1])) return 'Blazers';

  return m[1].split(' ').pop();

}


function readFields(r, p) {

  return {

    subject: field(r, p, 'Subject'), name: field(r, p, 'Player name'), bet: field(r, p, 'Bet'),

    ou: field(r, p, 'Over/Under'), line: field(r, p, 'Line'), period: field(r, p, 'Period'),

    details: field(r, p, 'Details'), sport: field(r, p, 'Sport'), team: field(r, p, 'Team'), opponent: field(r, p, 'Opponent')

  };

}


function describe(f, ctx) {

  let who = f.subject;

  if (who === 'Opponent') who = ctx.opp;

  else if (who === 'Opponent player' || who === 'Other player') who = f.name || who;

  else if (who === 'Game') who = ctx.matchup;

  if (!who) who = f.name || '';

  const ou = f.ou && f.ou !== 'None' ? f.ou : '';

  const line = num(f.line);

  const per = f.period && f.period !== 'Full game' ? f.period + ' ' : '';

  let s;

  switch (f.bet) {

    case 'Spread': s = `${who} ${line == null ? '' : signed(line)}`; break;

    case 'Moneyline': s = `${who} ML`; break;

    case 'Total': s = `${who} ${ou} ${line ?? ''}`; break;

    case 'Team total': s = `${who} TT ${ou} ${line ?? ''}`; break;

    case 'Other': case '': s = f.details || who; break;

    default: s = ou === 'Yes' || line == null ? `${who} ${f.bet}` : `${who} ${ou} ${line} ${f.bet}`;

  }

  s = (per + s).replace(/\s+/g, ' ').trim();

  if (f.bet && f.bet !== 'Other' && f.details) s += ` (${f.details})`;

  return s;

}


function profitFor(result, odds) {

  if (result === 'loss') return -STAKE;

  if (result !== 'win' || odds == null) return 0;

  return odds > 0 ? STAKE * odds / 100 : STAKE * 100 / Math.abs(odds);

}


function readPick(r, seg, ctx) {

    text = describe(f, { opp: f.opponent, matchup: f.opponent ? `${f.team}/${f.opponent}` : f.team });

    if ((f.bet === 'Spread' || f.bet === 'Moneyline') && f.opponent) text += ` vs ${f.opponent}`;

  } else if (p === 'OOT Prop') {

    const f = readFields(r, p);

    sport = f.sport; f.subject = '';

    text = describe(f, ctx);

  } else {

    text = describe(readFields(r, p), ctx);

  }

  const odds = num(field(r, p, 'Odds'));

  const boost = num(field(r, p, 'Boosted odds'));

  const raw = g(r, 'Result: ' + p).toLowerCase();

  return { seg, text, sport, odds, boost, eff, result,

    pts: result === 'win' ? seg.pts : 0, profit: profitFor(result, eff) };

}


function buildGames(rows) {

  const byGame = new Map();

  rows.forEach(r => { const game = g(r, 'Game'); if (game) byGame.set(game, r); }); // latest row wins

  const out = [];

  byGame.forEach((r, game) => {

    const opp = oppNick(game);

    const ctx = { opp, matchup: `Knicks/${opp}` };

    const picks = SEGMENTS.map(seg => readPick(r, seg, ctx));

    const graded = picks.filter(p => ['win', 'loss', 'push', 'void'].includes(p.result)).length;

    const idx = GAMES.indexOf(game);

    out.push({ game, picker: g(r, 'Name'), picks, graded, idx: idx < 0 ? 999 : idx,

      pts: picks.reduce((a, p) => a + p.pts, 0), profit: picks.reduce((a, p) => a + p.profit, 0) });

  });

  return out.sort((a, b) => a.idx - b.idx);

}

function blank() { return { games: 0, pts: 0, w: 0, l: 0, p: 0, profit: 0 }; }

function add(s, pk) {

  if (pk.result === 'win') s.w++; else if (pk.result === 'loss') s.l++; else if (pk.result === 'push' || pk.result === 'void') s.p++; else return;

  s.pts += pk.pts; s.profit += pk.profit;

}

  });

  return { players, crew, seg };

}

        <td class="num"><strong>${s.pts}</strong></td><td class="num">${s.games}</td>

        <td class="num">${s.games ? (s.pts / s.games).toFixed(1) : '–'}</td>

        <td class="num">${s.w}-${s.l}-${s.p}</td><td class="num">${pct(s.w, dec)}</td>

        <td class="num ${cls(s.profit)}">${money(s.profit)}</td>

        <td class="num ${cls(s.profit)}">${dec ? Math.round((s.profit / (dec * STAKE)) * 100) + '%' : '–'}</td></tr>`;

    }).join('') + '</tbody>';


  $('segments').innerHTML =

    `<thead><tr><th>Segment</th>${names.map(n => `<th class="num">${esc(n)}</th>`).join('')}<th class="num">Crew</th><th class="num">Hit %</th><th class="num">Profit</th></tr></thead><tbody>` +

    SEGMENTS.map((sg, i) => {

      const c = seg[i].crew;

      return `<tr><td class="segcell"><em>${sg.pts}</em>${esc(sg.label)}</td>` +

        names.map(n => { const s = seg[i].by[n] || blank(); return `<td class="num">${s.w + s.l + s.p ? `${s.w}-${s.l}${s.p ? '-' + s.p : ''}` : '–'}</td>`; }).join('') +

        `<td class="num">${c.w}-${c.l}${c.p ? '-' + c.p : ''}</td><td class="num">${pct(c.w, c.w + c.l)}</td><td class="num ${cls(c.profit)}">${money(c.profit)}</td></tr>`;

    }).join('') + '</tbody>';


  renderChart(names);

  renderFilters(names);

  renderLog();

}


function renderChart(names) {

  const graded = state.games.filter(gm => gm.graded);

  if (!graded.length) { $('chart').innerHTML = '<div class="empty-msg">The chart starts once the first game is graded.</div>'; $('legend').innerHTML = ''; return; }

  const W = 800, H = 280, L = 40, R = 16, T = 14, B = 34;

  const totals = Object.fromEntries(names.map(n => [n, [0]]));

  graded.forEach(gm => names.forEach(n => totals[n].push(totals[n].at(-1) + (gm.picker === n ? gm.pts : 0))));

  const maxY = Math.max(5, ...names.map(n => totals[n].at(-1)));

  const steps = graded.length;

  const x = i => L + (W - L - R) * (i / Math.max(steps, 1));

  const y = v => T + (H - T - B) * (1 - v / maxY);

  const tick = Math.max(1, Math.ceil(maxY / 5 / 5) * 5);

  let svg = `<svg viewBox="0 0 ${W} ${H}" role="img" aria-label="Cumulative points by picker">`;

  for (let v = 0; v <= maxY; v += tick) {

    svg += `<line x1="${L}" x2="${W - R}" y1="${y(v)}" y2="${y(v)}" style="stroke:var(--rule)"/>` +

           `<text x="${L - 8}" y="${y(v) + 4}" text-anchor="end" style="fill:var(--mute);font:12px var(--body)">${v}</text>`;

  }

  const every = Math.ceil(steps / 8);

  graded.forEach((gm, i) => {

    if (i % every === 0 || i === steps - 1)

      svg += `<text x="${x(i + 1)}" y="${H - 10}" text-anchor="middle" style="fill:var(--mute);font:12px var(--body)">${esc(gm.game.split(' ')[0])}</text>`;

  });

  names.forEach(n => {

    const pts = totals[n].map((v, i) => `${x(i).toFixed(1)},${y(v).toFixed(1)}`).join(' ');

    svg += `<polyline points="${pts}" style="fill:none;stroke:${COLORS[n] || '#888'};stroke-width:3;stroke-linejoin:round${DASHED.includes(n) ? ';stroke-dasharray:7 5' : ''}"/>`;

  });

  $('chart').innerHTML = svg + '</svg>';

  $('legend').innerHTML = names.map(n => `<span><span class="dot" style="background:${COLORS[n] || '#888'}"></span>${esc(n)} ${totals[n].at(-1)}</span>`).join('');

}


function renderFilters(names) {

  const sel = $('fPicker');

  sel.innerHTML = ['All', ...names].map(n => `<option value="${esc(n)}"${n === state.picker ? ' selected' : ''}>${n === 'All' ? 'All pickers' : esc(n)}</option>`).join('');

}


const RESULT_LABEL = { win: 'Win', loss: 'Loss', push: 'Push', void: 'Void', pending: 'Pending' };


function renderLog() {

  const q = state.q.toLowerCase();

  const list = state.games.slice().reverse().filter(gm =>

    (state.picker === 'All' || gm.picker === state.picker) &&

    (!q || (gm.game + ' ' + gm.picker + ' ' + gm.picks.map(p => p.text + ' ' + p.sport).join(' ')).toLowerCase().includes(q)));

  if (!list.length) { $('log').innerHTML = '<div class="empty-msg">No games match. Clear the search or pick another picker.</div>'; return; }

  $('log').innerHTML = list.map(gm => {

    const meter = gm.picks.map(p => `<div class="${p.result}" style="flex:${p.seg.pts}" title="${esc(p.seg.label)}: ${RESULT_LABEL[p.result] || 'Skipped'}">${'<i></i>'.repeat(p.seg.pts)}</div>`).join('');

    const rows = gm.picks.map(p => {

      if (p.result === 'empty') return `<li><span class="seg"><em>${p.seg.pts}</em>${esc(p.seg.label)}</span><span class="bet" style="color:var(--mute)">No pick</span><span></span><span></span></li>`;

      const odds = p.boost != null && p.odds != null ? `<s>${signed(p.odds)}</s>${signed(p.boost)}` : (p.eff != null ? signed(p.eff) : '');

      return `<li><span class="seg"><em>${p.seg.pts}</em>${esc(p.seg.label)}</span>

        <span class="bet">${esc(p.text)}${p.sport ? `<small>${esc(p.sport)}</small>` : ''}</span>

        <span class="odds">${odds}</span><span class="res ${p.result}">${RESULT_LABEL[p.result]}</span></li>`;

    }).join('');

    const status = gm.graded === SEGMENTS.length ? '' : gm.graded ? ' so far' : ' (not graded)';

    return `<article class="game"><div class="ghead"><h3>${esc(gm.game)}</h3>

      <div class="meta">${esc(gm.picker || 'Unknown')} <b>${gm.pts} of ${MAX_PTS} pts</b>${status} <b class="${cls(gm.profit)}" style="color:inherit">${money(gm.profit)}</b></div></div>

      <div class="meter" aria-hidden="true">${meter}</div><ul class="picks">${rows}</ul></article>`;

  }).join('');

}

/* ---------- load ---------- */

const isDemo = () => !CSV_URL && (!SHEET_ID || SHEET_ID.startsWith('PASTE'));


async function load() {

  $('updated').textContent = 'Loading…';

  $('notice').innerHTML = '';

  try {

    let rows;

    if (isDemo()) {

      rows = demoRows();

      $('notice').innerHTML = '<p class="note">Showing sample picks. Paste the responses sheet ID into SHEET_ID at the top of the script to go live.</p>';

    } else {

      const url = CSV_URL || `https://docs.google.com/spreadsheets/d/${SHEET_ID}/gviz/tq?tqx=out:csv&headers=1&sheet=${encodeURIComponent(TAB)}`;

      const res = await fetch(url, { cache: 'no-store' });

      if (!res.ok) throw new Error(`The sheet answered with error ${res.status}.`);

      const text = await res.text();

      if (/^\s*</.test(text)) throw new Error('The sheet sent back a sign-in page instead of data.');

      rows = toObjects(parseCSV(text));

    }

    state.games = buildGames(rows);

    render();

    $('updated').textContent = 'Updated ' + new Date().toLocaleTimeString([], { hour: 'numeric', minute: '2-digit' });

  } catch (e) {

    $('updated').textContent = 'Not loaded';

    $('notice').innerHTML = `<p class="note err">Couldn’t load the picks. ${esc(e.message)} Check that SHEET_ID is the responses sheet’s ID and that the sheet is shared as “Anyone with the link: Viewer.”</p>`;

  }

}


document.querySelectorAll('.stake').forEach(el => el.textContent = STAKE);

$('refresh').addEventListener('click', load);

$('fPicker').addEventListener('change', e => { state.picker = e.target.value; renderLog(); });

$('fSearch').addEventListener('input', e => { state.q = e.target.value; renderLog(); });

load();

setInterval(load, 5 * 60 * 1000);
